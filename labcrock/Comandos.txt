Descargar

Docker Desktop

verificar en powershell si esta instalado

docker --version

comprobar que no de error nada
docker ps

descargar la imagen de cockroach
docker pull cockroachdb/cockroach:latest

crear la carpeta para la practica
mkdir LabCockroach
cd LabCockroach

dentro de esa carpeta crear este archivo
docker-compose.yml


dentro de ese archivo copiar este bloque
version: '3.8'

services:
  roach1:
    image: cockroachdb/cockroach:latest
    container_name: roach1
    hostname: roach1
    command: start --insecure --join=roach1,roach2,roach3
    ports:
      - "26257:26257"
      - "8080:8080"

  roach2:
    image: cockroachdb/cockroach:latest
    container_name: roach2
    hostname: roach2
    command: start --insecure --join=roach1,roach2,roach3

  roach3:
    image: cockroachdb/cockroach:latest
    container_name: roach3
    hostname: roach3
    command: start --insecure --join=roach1,roach2,roach3


iniciar Docker compose
docker compose up -d

verificar
docker ps

iniciar el anillo
docker exec roach1 ./cockroach init --insecure

esta es la pagina web donde también se puede ver el estado de los nodos
http://localhost:8080


Entrar al SQL

docker exec -it roach1 ./cockroach sql --insecure

Crear la base de datos FarmaEC

CREATE DATABASE farmaec;
USE farmaec;

Crear tabla clientes

CREATE TABLE clientes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    cedula STRING,
    nombre STRING,
    ciudad STRING,
    telefono STRING,
    total_compras DECIMAL
);

SHOW RANGES FROM TABLE clientes;

BEGIN;
UPDATE clientes
SET total_compras = total_compras + 50
